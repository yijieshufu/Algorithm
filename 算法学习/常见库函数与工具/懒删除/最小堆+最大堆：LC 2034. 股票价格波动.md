# 题目
实时接收 `(时间戳, 价格)`。  
支持更新（旧时间戳新价格）、查询最新价格、查询历史最高/最低价格。
# 分析

用时间来作为一个键

找 `maximum`时 判断一下 时间是不是对的  
	堆里面的时间 和 字典里面的时间是否一致  
		堆是按照 值来排序的 
# 代码
```python
import heapq
class StockPrice:

    def __init__(self):
        self.d={} # key 记录时间, val:价格
        self.hi = [] # 最小堆
        self.ha = [] # 最大堆
        self.m = 0 # 最新时间

    def update(self, timestamp: int, price: int) -> None:
        t = timestamp
        p = price
        self.d[t] = p
        self.m = max(self.m,t) # 记录最新时间
        heapq.heappush(self.hi,(p,t))
        heapq.heappush(self.ha,(-p,t))
    def current(self) -> int:
        return self.d[self.m]
    def maximum(self) -> int:
        while -self.ha[0][0] != self.d[self.ha[0][1]]: # 跟最新时间的价格对不上
            heapq.heappop(self.ha)
        return -self.ha[0][0]

    def minimum(self) -> int:
        while self.hi[0][0] != self.d[self.hi[0][1]]: # 跟最新时间的价格对不上
            heapq.heappop(self.hi)
        return self.hi[0][0]

```
